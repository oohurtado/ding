

users
-----
id              int PK AI
first_name      varchar(50)
last_name       varchar(50)
password_hash   varchar(255)
email           varchar(100) UNIQUE


sections
--------
id              int PK AI
user_id         int FK -> users.id
name            varchar(50)
position        int

UNIQUE(user_id, name)
UNIQUE(user_id, position)


categories
----------
id              int PK AI
section_id      int FK -> sections.id
name            varchar(50)
budget          decimal(12,2)
is_service      bool
position        int

UNIQUE(section_id, name)
UNIQUE(section_id, position)


expenses
--------
id              int PK AI
category_id     int FK -> categories.id
description     varchar(250) NULL
date            datetime
amount          decimal(12,2)


tags
----
id              int PK AI
user_id         int FK -> users.id
name            varchar(25)

UNIQUE(user_id, name)


expense_tags
------------
id              int PK AI
expense_id      int FK -> expenses.id
tag_id          int FK -> tags.id

PK(expense_id, tag_id)
